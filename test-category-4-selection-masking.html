💡 How to Use Category 4 Features

AI-Powered Selection

Call functions from JavaScript console or add UI buttons:

// Select subject
aiSelectSubject();

// Select sky
aiSelectSky();

Color Range Selection

// Select by color
colorRangeSelection('#ff0000', 40);

// Skin tone selection
colorRangeSelection(null, 40, false, true);

Luminosity Masks

// Create highlights mask
createLuminosityMask('highlights');

// Custom range
createLuminosityMask('custom', 100, 200, 15);

Select and Mask Workspace

// Open workspace (after making selection)
openSelectAndMaskWorkspace();

// Refine edges
refineSelectionEdges(true, 5);

Gradient Masks

// Linear gradient mask
createGradientMask('linear', 0, 0, 500, 500);

// Radial gradient mask
createGradientMask('radial', 250, 250, 400, 400);

Selection Algebra

// Set algebra mode
state.selection.algebra = 'add';  // or 'subtract', 'intersect', 'xor'

// Make new selection to combine
// Selection will automatically use algebra mode
